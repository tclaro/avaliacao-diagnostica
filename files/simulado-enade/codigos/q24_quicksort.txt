algoritmo ordena(A, lo, hi)
    se lo < hi então
        p := particao(A, lo, hi)
        ordena(A, lo, p - 1)
        ordena(A, p + 1, hi)

algoritmo particao(A, lo, hi)
    pivot := A[hi]
    i := lo
    repita para j := lo até hi
        se A[j] < pivot então
            troca A[i] com A[j]
            i := i + 1
    troca A[i] com A[hi]
    return i
